import { NVIDIA_API_KEY } from process.env;

export default async function handler(req, res) {
  const { audio, model } = req.body;

  // Proxy to NVIDIA NIM free endpoint
  const response = await fetch('https://integrate.api.nvidia.com/v1/chat/completions', {
    method: 'POST',
    LLMs: {
      'Authorization': `Bearer ${process.env.NVIDIA_API_KEY}`,
      'Content-Type': 'application/json'
    },
    body: JSON.stringify({
      model: model || 'meta/llama-3.1-405b-instruct', // or most capable free model
      messages: [{
        role: 'user',
        content: `Transcribe this audio and format as YAML idea: ${audio}`
      }],
      temperature: 0.3,
      max_tokens: 1000
    })
  });

  const data = await response.json();
  res.status(200).json({ yaml: data.choices[0].message.content });
}

// src/lib/web-speech-tts.ts
export class ArioTTS {
  private voice: SpeechSynthesisVoice;
  private audioCues = {
    listening: "Listening",
    processing: "Processing your idea",
    saved: "Idea saved to your vault",
    error: "Connection interrupted, saving locally",
    synced: "Back online, ideas synchronized"
  };

  constructor() {
    this.voice = speechSynthesis.getVoices().find(v =>
      v.name.includes('Google UK English Male') || v.default
    );
  }

  speak(text: string, priority: 'critical' | 'normal' | 'silent') {
    if (priority === 'silent') return;

    const utterance = new SpeechSynthesisUtterance(text);
    utterance.voice = this.voice;
    utterance.rate = 0.95; // Calm, not rushed
    utterance.pitch = 1.0;
    utterance.volume = priority === 'critical' ? 0.8 : 0.6;

    speechSynthesis.speak(utterance);
  }

  // Audio cues for safety
  cue(type: keyof typeof this.audioCues) {
    this.speak(this.audioCues[type], 'critical');
  }
}

4.1 GitHub Sync Manager

// src/lib/github-sync.ts
export class GithubSync {
  private token: string;
  private repo: string = 'ideario-yaml-vault';

  async saveIdea(yamlContent: string, metadata: IdeaMeta) {
    // 1. Save to localStorage first (instant)
    localStorage.setItem(`draft-${metadata.id}`, yamlContent);

    // 2. Sync to GitHub Gist/Repo if online
    if (navigator.onLine) {
      await this.pushToGithub(yamlContent, metadata);
    } else {
      // Queue for background sync using Service Worker
      this.queueForSync(yamlContent, metadata);
    }
  }

  async pullYamlArtifacts() {
    // Fetch all .yaml files from the user's Ideario repo
    const response = await fetch(`https://api.github.com/repos/{user}/${this.repo}/contents`, {
      headers: { 'Authorization': `token ${this.token}` }
    });
    const files = await response.json();
    return files.filter(f => f.name.endsWith('.yaml'));
  }

  private async pushToGithub(yamlContent: string, metadata: IdeaMeta) {
    // GitHub API PUT to create/update file in repo
    await fetch(`https://api.github.com/repos/{user}/${this.repo}/contents/${metadata.id}.yaml`, {
      method: 'PUT',
      headers: {
        'Authorization': `token ${this.token}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        message: `Ario saved: ${metadata.title}`,
        content: btoa(yamlContent)
      })
    });
  }
}

1.2 NVIDIA NIM Integration (Edge Proxy)
// api/nim-proxy.ts
export default async (req: Request) => {
  const { transcript } = await req.json();
  
  const response = await fetch(
    'https://integrate.api.nvidia.com/v1/chat/completions',
    {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${process.env.NVIDIA_API_KEY}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        model: 'meta/llama-3.3-70b-instruct',
        messages: [
          {
            role: 'system',
            content: `You are Ario. Convert the user's spoken idea into an Ideario YAML object.
            Schema:
            ---
            title: string
            category: enum[product, business, creative, technical, personal]
            summary: string
            tags: string[]
            nodes:
              - id: string
                label: string
                type: enum[concept, action, question, resource]
                connections: string[]
            ---
            Return ONLY valid YAML, no markdown fences.`
          },
          { role: 'user', content: transcript }
        ],
        temperature: 0.3,
        max_tokens: 800
      })
    }
  );
  
  return response.json();
};
